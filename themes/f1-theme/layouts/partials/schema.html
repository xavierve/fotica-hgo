WebSite ---------- */}} {{- $graph = $graph | append (dict "@type" "WebSite" "@id" $websiteId "url" $base "name" site.Title "inLanguage" "es-ES" "publisher" (dict "@id" $orgId) ) -}} {{/* ---------- 3b. Equipo (Person), solo en la página que los presenta ---------- No se repite en cada página (serían 7 Person con bio completa en el JSON-LD de TODO el sitio): solo en /nosotros/, que es donde vive el contenido real de equipo. Nodo de la página actual ---------- */}} {{- $pageType := $schema.type | default (cond .IsHome "WebPage" "WebPage") -}} {{- $pageId := printf "%s#%s" .Permalink (lower $pageType) -}} {{- $page := dict "@type" $pageType "@id" $pageId "name" (or .Params.seo.title .Params.title .Title) "url" .Permalink "inLanguage" "es-ES" -}} {{- with or .Params.seo.description .Params.description }}{{ $page = merge $page (dict "description" .) }}{{ end -}} {{- if eq $pageType "Service" -}} {{- $page = merge $page (dict "provider" (dict "@id" $orgId) "areaServed" (dict "@type" "Place" "name" ($siteData.business.areaServed | default site.Title)) ) -}} {{- with .Params.schema.serviceType }}{{ $page = merge $page (dict "serviceType" .) }}{{ end -}} {{- else -}} {{- $page = merge $page (dict "isPartOf" (dict "@id" $websiteId)) -}} {{- end -}} {{- $graph = $graph | append $page -}} {{/* ---------- 5.
